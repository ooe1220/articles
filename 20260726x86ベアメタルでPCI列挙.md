# 実装

`kernel.c`以外のソースは以下の記事内にあります。
[DVDから直接起動してC言語を動かす手順](https://qiita.com/earthen94/items/bc0817a7035c0a940a11)

```bash
rmdir tmp
mkdir tmp

nasm -f bin boot.asm -o tmp/boot.bin

gcc -m32 -ffreestanding -c kernel.c -o tmp/kernel.o
nasm -f elf32 to_32bit.asm -o tmp/to_32bit.o
ld -m elf_i386 -T linker.ld -o tmp/kernel.elf tmp/to_32bit.o tmp/kernel.o
objcopy -O binary tmp/kernel.elf tmp/kernel.bin
cat tmp/boot.bin tmp/kernel.bin > tmp/disk.img

rmdir iso
rm os.iso

mkdir iso
cp tmp/disk.img iso/

xorriso -as mkisofs \
  -o os.iso \
  -b disk.img \
  -c boot.cat \
  -no-emul-boot \
  -boot-load-size 4 \
  iso

qemu-system-i386 -cdrom os.iso -m 512 -boot d

sudo dvd+rw-format -blank=fast /dev/sr0
sudo growisofs -dvd-compat -Z /dev/sr0=os.iso
```

```kernel.c
#define PCI_CONFIG_ADDRESS  0xCF8
#define PCI_CONFIG_DATA     0xCFC

/* I/O */
static inline void outl(unsigned int port, unsigned int value)
{
    __asm__ volatile ("outl %0, %1" : : "a"(value), "Nd"(port));
}

static inline unsigned int inl(unsigned int port)
{
    unsigned int ret;
    __asm__ volatile ("inl %1, %0" : "=a"(ret) : "Nd"(port));
    return ret;
}

/*
 * どのPCI機器のどのレジスタを読みたいかを、PCI仕様書が定めた32bitの形式に変換
 * bus:    0-255
 * device: 0-31
 * func:   0-7
 * reg:    0-63
 */
static unsigned int pci_config_addr(unsigned char bus,
                                     unsigned char device,
                                     unsigned char func,
                                     unsigned char reg)
{
    return (1 << 31) |
           ((bus & 0xFF) << 16) |
           ((device & 0x1F) << 11) |
           ((func & 0x07) << 8) |
           ((reg & 0x3F) << 2);
}

/* PCIデバイスの設定情報を1つの32bit値として読み出す */
static unsigned int pci_read32(unsigned char bus,
                                unsigned char device,
                                unsigned char func,
                                unsigned char reg)
{
    // 0xCF8にどの機器のどのレジスタを読みたいか」を指定するアドレス値を書き込む
    outl(PCI_CONFIG_ADDRESS, pci_config_addr(bus, device, func, reg));
    
    // 0xCFCから32bit値を読み出す
    return inl(PCI_CONFIG_DATA);
}

/* Vendor ID及びDevice IDを取り出す */
static void pci_read_vendor_device(unsigned char bus,
                                    unsigned char device,
                                    unsigned short *vendor,
                                    unsigned short *device_id)
{
    unsigned int val = pci_read32(bus, device, 0, 0);
    *vendor     = val & 0xFFFF;
    *device_id  = (val >> 16) & 0xFFFF;
}

void putc(char c) {
    static int pos = 0;   // 0〜1999 の範囲でカーソル位置を管理

    if (c == '\n') {
        /* 次の行の先頭に移動 */
        int row = pos / 80;
        row++;                 // 1行進める
        if (row >= 25) {       // 25行目以降はスクロールせず止める
            pos = 2000;
            return;
        }
        pos = row * 80;
        return;
    }

    if (pos >= 2000) {
        return;  // 画面末尾を超えたら書かない
    }

    volatile unsigned short* vram = (unsigned short*)0xB8000;
    vram[pos++] = 0x0F00 | c;
}

void puts(const char* s) {
    while (*s) putc(*s++);
}
/* 数値を16進数4桁で表示 (例: 0x1A2B) */
void put_hex4(unsigned short val)
{
    int i;
    putc('0');
    putc('x');

    /* 上位ビットから順に処理 */
    for (i = 12; i >= 0; i -= 4) {
        unsigned char digit = (val >> i) & 0xF;
        if (digit < 10)
            putc('0' + digit);
        else
            putc('A' + digit - 10);
    }
}

void kernel_main() {    
    unsigned char bus, device;
    unsigned short vendor, device_id;

    // 機器のみ。FUNTIONは一旦探索しない
    for (bus = 0; bus < 256; bus++) {
        for (device = 0; device < 32; device++) {

            pci_read_vendor_device(bus, device, &vendor, &device_id);

            if (vendor == 0xFFFF) {
                continue;   /* 機器が存在しない */
            }
 
            puts("vendor = ");           
            put_hex4(vendor);
            puts(" ");
            puts("device_id = ");           
            put_hex4(device_id);
            puts(" \n");

        }
    }

    while(1);
}
```
