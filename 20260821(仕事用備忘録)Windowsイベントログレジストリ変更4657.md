```test.c
#include <string.h>
#include <windows.h>
#include <stdio.h>

int main(void)
{
    HKEY hKey;
    const char *subkey = "Software\\TestAudit";
    const char *value = "Hello";

    LONG ret = RegCreateKeyExA(
        HKEY_CURRENT_USER,
        subkey,
        0,
        NULL,
        0,
        KEY_SET_VALUE,
        NULL,
        &hKey,
        NULL
    );

    if (ret != ERROR_SUCCESS) {
        printf("RegCreateKeyExA failed: %ld\n", ret);
        return 1;
    }

    ret = RegSetValueExA(
        hKey,
        "TestValue",
        0,
        REG_SZ,
        (const BYTE *)value,
        (DWORD)(strlen(value) + 1)
    );

    if (ret != ERROR_SUCCESS) {
        printf("RegSetValueExA failed: %ld\n", ret);
        RegCloseKey(hKey);
        return 1;
    }

    RegCloseKey(hKey);

    printf("Registry value written.\n");
    return 0;
}
```
